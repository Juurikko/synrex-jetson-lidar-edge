cmake_minimum_required(VERSION 3.20)
project(synrex_native_byom VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(LAZPERF_SOURCE_DIR "" CACHE PATH "External laz-perf 3.4.0 source checkout")
if(NOT EXISTS "${LAZPERF_SOURCE_DIR}/cpp/lazperf/readers.hpp")
  message(FATAL_ERROR "Set LAZPERF_SOURCE_DIR to an external laz-perf 3.4.0 checkout")
endif()
# Compile only the documented codec library; no upstream datasets/tests/tools are fetched.
file(GLOB LAZPERF_SOURCES CONFIGURE_DEPENDS
  "${LAZPERF_SOURCE_DIR}/cpp/lazperf/*.cpp"
  "${LAZPERF_SOURCE_DIR}/cpp/lazperf/detail/*.cpp")
add_library(lazperf_review STATIC ${LAZPERF_SOURCES})
target_include_directories(lazperf_review SYSTEM PUBLIC "${LAZPERF_SOURCE_DIR}/cpp")
target_compile_definitions(lazperf_review PUBLIC LAZPERF_VENDORED)
add_executable(synrex-laz-labels src/preserve_laz.cpp)
target_link_libraries(synrex-laz-labels PRIVATE lazperf_review)
target_compile_options(synrex-laz-labels PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(SYNREX_SANITIZE "Address/undefined-behavior sanitizer build" OFF)
if(SYNREX_SANITIZE)
  foreach(t lazperf_review synrex-laz-labels)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endforeach()
  target_link_options(synrex-laz-labels PRIVATE -fsanitize=address,undefined)
endif()

# Existing model-free writer can still be built with its original dependencies only.
option(SYNREX_BUILD_BYOM "Build the native BYOM runtime" ON)
if(SYNREX_BUILD_BYOM)
# Package validation, geometry and the original LAZ writer build without a GPU SDK.
set(NLOHMANN_JSON_INCLUDE_DIR "" CACHE PATH "Directory containing nlohmann/json.hpp (3.11.3)")
find_path(SYNREX_JSON_INCLUDE nlohmann/json.hpp HINTS "${NLOHMANN_JSON_INCLUDE_DIR}")
if(NOT SYNREX_JSON_INCLUDE)
  message(FATAL_ERROR "Set NLOHMANN_JSON_INCLUDE_DIR to nlohmann/json 3.11.3 single_include")
endif()
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
add_library(synrex_core STATIC native/package.cpp native/geometry.cpp src/preserve_laz.cpp)
target_include_directories(synrex_core PUBLIC native/include)
target_include_directories(synrex_core SYSTEM PUBLIC "${SYNREX_JSON_INCLUDE}")
target_compile_definitions(synrex_core PRIVATE SYNREX_LAZ_LIBRARY _GNU_SOURCE)
target_link_libraries(synrex_core PUBLIC lazperf_review OpenSSL::Crypto)
target_compile_options(synrex_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(SYNREX_ENABLE_CUDA "Build genuine CUDA/TensorRT backend (TensorRT 10.x)" OFF)
if(SYNREX_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(TENSORRT_ROOT "" CACHE PATH "TensorRT SDK root")
  find_path(TENSORRT_INCLUDE NvInfer.h HINTS "${TENSORRT_ROOT}/include" /usr/include/aarch64-linux-gnu)
  find_library(TENSORRT_LIBRARY nvinfer HINTS "${TENSORRT_ROOT}/lib" "${TENSORRT_ROOT}/lib64")
  find_library(ONNXPARSER_LIBRARY nvonnxparser HINTS "${TENSORRT_ROOT}/lib" "${TENSORRT_ROOT}/lib64")
  if(NOT TENSORRT_INCLUDE OR NOT TENSORRT_LIBRARY OR NOT ONNXPARSER_LIBRARY)
    message(FATAL_ERROR "TensorRT 10.x headers, nvinfer and nvonnxparser required")
  endif()
  add_library(synrex_backend OBJECT native/tensorrt.cpp native/kernels.cu)
  target_include_directories(synrex_backend SYSTEM PRIVATE "${TENSORRT_INCLUDE}")
  target_link_libraries(synrex_backend PUBLIC synrex_core CUDA::cudart "${TENSORRT_LIBRARY}" "${ONNXPARSER_LIBRARY}")
  set_target_properties(synrex_backend PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(synrex_backend PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Werror>)
else()
  add_library(synrex_backend OBJECT native/no_gpu.cpp)
  target_link_libraries(synrex_backend PUBLIC synrex_core)
endif()
add_executable(synrex-native native/main.cpp)
target_link_libraries(synrex-native PRIVATE synrex_backend)
target_compile_options(synrex-native PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(synrex-core-tests tests/native/core.cpp)
  target_link_libraries(synrex-core-tests PRIVATE synrex_backend)
  target_compile_options(synrex-core-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME dense_voxel_contract COMMAND synrex-core-tests "${PROJECT_SOURCE_DIR}/examples/dense_voxel.json")
endif()
if(SYNREX_SANITIZE)
  foreach(t synrex_core synrex_backend synrex-native synrex-core-tests)
    if(TARGET ${t})
      target_compile_options(${t} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fsanitize=address,undefined;-fno-omit-frame-pointer>)
      target_link_options(${t} PUBLIC -fsanitize=address,undefined)
    endif()
  endforeach()
endif()
include(GNUInstallDirs)
install(TARGETS synrex-native synrex-laz-labels RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES LICENSE LICENSE_SCOPE.json THIRD_PARTY_NOTICES.md DESTINATION "${CMAKE_INSTALL_DATADIR}/synrex")
install(DIRECTORY LICENSES/ DESTINATION "${CMAKE_INSTALL_DATADIR}/synrex/licenses")
install(DIRECTORY docs/byom/ DESTINATION "${CMAKE_INSTALL_DATADIR}/synrex/docs")
install(DIRECTORY examples/ DESTINATION "${CMAKE_INSTALL_DATADIR}/synrex/examples")
else()
  install(TARGETS synrex-laz-labels RUNTIME DESTINATION bin)
endif()
