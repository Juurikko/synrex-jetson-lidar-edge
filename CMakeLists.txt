cmake_minimum_required(VERSION 3.20)

set(SYNREX_PORT_TARGET "" CACHE STRING "Optional native ARM64 target: orin or thor")
if(SYNREX_PORT_TARGET)
  if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    message(FATAL_ERROR "Orin/Thor qualification configurations require native ARM64; RTX is not target evidence")
  endif()
  if(NOT SYNREX_PORT_TARGET MATCHES "^(orin|thor)$" OR NOT SYNREX_ENABLE_CUDA)
    message(FATAL_ERROR "Unknown port target or missing CUDA backend")
  endif()
  if((SYNREX_PORT_TARGET STREQUAL "orin" AND NOT CMAKE_CUDA_ARCHITECTURES STREQUAL "87") OR
     (SYNREX_PORT_TARGET STREQUAL "thor" AND NOT CMAKE_CUDA_ARCHITECTURES STREQUAL "110"))
    message(FATAL_ERROR "Port target must use its physical SM; no RTX engine portability")
  endif()
endif()

set(LAZPERF_SOURCE_DIR "" CACHE PATH "External laz-perf 3.4.0 source checkout")
if(NOT EXISTS "${LAZPERF_SOURCE_DIR}/cpp/lazperf/readers.hpp")
  message(FATAL_ERROR "Set LAZPERF_SOURCE_DIR to an external laz-perf 3.4.0 checkout")
endif()
# Compile only the documented codec library; no upstream datasets/tests/tools are fetched.
file(GLOB LAZPERF_SOURCES CONFIGURE_DEPENDS
  "${LAZPERF_SOURCE_DIR}/cpp/lazperf/*.cpp"
  "${LAZPERF_SOURCE_DIR}/cpp/lazperf/detail/*.cpp")
add_library(lazperf_review STATIC ${LAZPERF_SOURCES})
target_include_directories(lazperf_review SYSTEM PUBLIC "${LAZPERF_SOURCE_DIR}/cpp")
target_compile_definitions(lazperf_review PUBLIC LAZPERF_VENDORED)
add_executable(synrex-laz-labels src/preserve_laz.cpp)
target_link_libraries(synrex-laz-labels PRIVATE lazperf_review)
target_compile_options(synrex-laz-labels PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(SYNREX_SANITIZE "Address/undefined-behavior sanitizer build" OFF)
if(SYNREX_SANITIZE)
  foreach(t lazperf_review synrex-laz-labels)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endforeach()
  target_link_options(synrex-laz-labels PRIVATE -fsanitize=address,undefined)
endif()

option(SYNREX_ENABLE_CUDA "Build genuine CUDA/TensorRT backend (TensorRT 10.x)" OFF)
if(SYNREX_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(TENSORRT_ROOT "" CACHE PATH "TensorRT SDK root")
  find_path(TENSORRT_INCLUDE NvInfer.h HINTS "${TENSORRT_ROOT}/include" /usr/include/aarch64-linux-gnu)
  find_library(TENSORRT_LIBRARY nvinfer HINTS "${TENSORRT_ROOT}/lib" "${TENSORRT_ROOT}/lib64")
  find_library(ONNXPARSER_LIBRARY nvonnxparser HINTS "${TENSORRT_ROOT}/lib" "${TENSORRT_ROOT}/lib64")
  if(NOT TENSORRT_INCLUDE OR NOT TENSORRT_LIBRARY OR NOT ONNXPARSER_LIBRARY)
    message(FATAL_ERROR "TensorRT 10.x headers, nvinfer and nvonnxparser required")
  endif()
  add_library(synrex_backend OBJECT native/tensorrt.cpp native/kernels.cu)
  target_include_directories(synrex_backend SYSTEM PRIVATE "${TENSORRT_INCLUDE}")
  target_link_libraries(synrex_backend PUBLIC synrex_core CUDA::cudart "${TENSORRT_LIBRARY}" "${ONNXPARSER_LIBRARY}")
  set_target_properties(synrex_backend PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
  target_compile_options(synrex_backend PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  target_compile_options(synrex_backend PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Werror>)
else()
  add_library(synrex_backend OBJECT native/no_gpu.cpp)
  target_link_libraries(synrex_backend PUBLIC synrex_core)
endif()
add_executable(synrex-native native/main.cpp)

# Propagate backend objects directly, including for API consumers that reference only parser symbols.
